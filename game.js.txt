// تهيئة محرك PlayCanvas لعرض ساحة المعركة ثلاثية الأبعاد
const canvas = document.getElementById('application-canvas');
const app = new pc.Application(canvas, {
    mouseInput: new pc.MouseInput(canvas),
    touchInput: new pc.TouchInput(canvas)
});
app.setCanvasFillMode(pc.FILLMODE_FILL_WINDOW);
app.setCanvasResolution(pc.RESOLUTION_AUTO);

// إعداد المشهد التوليدي (الجزيرة وساحة المعركة)
app.on('update', function (dt) {
    // حركة دائرية خفيفة للكاميرا أو عناصر المعركة لإظهار الحيوية التوليدية
    if (window.islandEntity) {
        window.islandEntity.rotate(0, 10 * dt, 0);
    }
});

function init3DWorld() {
    // إنشاء الكاميرا
    const camera = new pc.Entity();
    camera.addComponent("camera", {
        clearColor: new pc.Color(0.05, 0.07, 0.1)
    });
    camera.setLocalPosition(0, 5, 10);
    camera.lookAt(0, 0, 0);
    app.root.addChild(camera);

    // إضافة إضاءة للجزيرة
    const light = new pc.Entity();
    light.addComponent("light", {
        type: "directional",
        color: new pc.Color(1, 1, 1),
        intensity: 1.5
    });
    light.setLocalEulerAngles(45, 30, 0);
    app.root.addChild(light);

    // إنشاء نموذج تجريبي يمثل الجزيرة / ساحة المعركة التوليدية
    const island = new pc.Entity();
    island.addComponent("render", {
        type: "box"
    });
    island.setLocalScale(4, 0.5, 4);
    app.root.addChild(island);
    window.islandEntity = island;

    app.start();
}

// تشغيل المحرك ثلاثي الأبعاد عند تحميل الصفحة
init3DWorld();

document.getElementById('event-btn').addEventListener('click', () => {
    const status = document.getElementById('status-display');
    const eventsList = [
        "حدث عاصفة رملية مفاجئة (تأثير بصرى متساوى للجميع)",
        "حدث إسقاط إمدادات تجميلية محايدة فى منتصف الخريطة",
        "حدث تعديل إضاءة الخريطة بنظام الذكاء الاصطناعي لزيادة الحماس"

    const randomEvent = eventsList[Math.floor(Math.random() * eventsList.length)];

    setTimeout(() => {
        status.innerText = `تم إطلاق الحدث بنجاح: ${randomEvent}`;
    }, 2000);
});

document.getElementById('gear-btn').addEventListener('click', () => {
    const gearInput = document.getElementById('gear-input').value;
    const status = document.getElementById('status-display');
    if(!gearInput.trim()) {
        status.innerText = "الرجاء كتابة وصف المظهر أو العتاد التجميلي أولاً!";
        return;
    }
    status.innerText = `جاري تجهيز التخصيص التجميلي المتوازن ("${gearInput}")...`;
    setTimeout(() => {
        status.innerText = `تمت إضافة المظهر المخصص بنجاح (ضمن حدود اللعب العادل الفير بلاى)!`;
    }, 2000);
});
